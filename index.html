 export async function intakeHandler(req: Request, res: Response) {
   const tenantId = String(req.headers["x-tenant-id"] || "");
   const deliveryId = String(req.headers["x-delivery-id"] || "");
   const rawBody = req.rawBody;
 
   if (!verifySignature(rawBody, req.headers["x-signature"] as string, secretFor(tenantId))) {
     return res.status(401).json({ error: "invalid signature" });
   }
 
+  // Fast path: avoid duplicate work on vendor retries
+  const seen = await redis.get(`delivery:${deliveryId}`);
+  if (seen) {
+    return res.status(200).json({ ok: true, deduped: true });
+  }
 
   const event = JSON.parse(rawBody.toString("utf8")) as WebhookEvent;
 
+  try {
     const caseId = await caseApi.create({
       tenantId,
       externalId: event.id,
       title: event.title,
       payload: event,
     });
+    await redis.set(`delivery:${deliveryId}`, caseId, "EX", 60); // 60s is enough for retries
+    return res.status(200).json({ ok: true, caseId });
+  } catch (err) {
+    logger.error("case.create", { err, tenantId, externalId: event.id });
+    // Always acknowledge so the vendor stops retrying noisy failures
+    return res.status(200).json({ ok: true });

 }
